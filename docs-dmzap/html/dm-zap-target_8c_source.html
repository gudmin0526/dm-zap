271 /*
272  * Process a BIO.
273  */
274 int dmzap_handle_bio(struct dmzap_target *dmzap,
275  struct dmzap_chunk_work *cw, struct bio *bio)
276 {
277  int ret;
278 
279  if (dmzap->dev->flags & DMZ_BDEV_DYING) {
280  ret = -EIO;
281  goto out;
282  }
283 
284  if (!bio_sectors(bio)) {
285  ret = DM_MAPIO_SUBMITTED;
286  goto out;
287  }
288 
289  /*
290  * Write may trigger a zone allocation. So make sure the
291  * allocation can succeed.
292  */
293  if (bio_op(bio) == REQ_OP_WRITE){
294  mutex_lock(&dmzap->reclaim_lock);
295  dmzap_schedule_reclaim(dmzap);
296  mutex_unlock(&dmzap->reclaim_lock);
297  dmzap->nr_user_written_sec += bio_sectors(bio);
298  }
299 
300  switch (bio_op(bio)) {
301  case REQ_OP_READ:
302  mutex_lock(&dmzap->map.map_lock);
303  ret = dmzap_conv_read(dmzap, bio);
304  mutex_unlock(&dmzap->map.map_lock);
305  break;
306  case REQ_OP_WRITE:
307  mutex_lock(&dmzap->map.map_lock);
308  ret = dmzap_conv_write(dmzap, bio);
309  mutex_unlock(&dmzap->map.map_lock);
310 
311  break;
312  case REQ_OP_DISCARD:
313  case REQ_OP_WRITE_ZEROES:
314  dmz_dev_debug(dmzap->dev, "Discard operation triggered");
315  ret = dmzap_handle_discard(dmzap, bio);
316  break;
317  default:
318  dmz_dev_err(dmzap->dev, "Ignoring unsupported BIO operation 0x%x",
319  bio_op(bio));
320  ret = -EIO;
321  }
322 
323  /* Upadate access time*/
324  dmzap_reclaim_bio_acc(dmzap);
325 
326  if(time_is_before_jiffies(dmzap->wa_print_time + DMZAP_WA_PERIOD)){
327  trace_printk("User written sectors: %lld, GC written sectors: %lld, vsm: %u, dev_c: %lld\n",
328  dmzap->nr_user_written_sec,
329  dmzap->nr_gc_written_sec,
330  dmzap->victim_selection_method,
331  dmzap->dev_capacity);
332  dmzap->wa_print_time = jiffies;
333  }
334 out:
335  dmzap_bio_endio(bio, errno_to_blk_status(ret));
336  return ret;
337 }

360 /*
361  * Chunk BIO work function.
362  */
363 static void dmzap_chunk_work_(struct work_struct *work)
364 {
365  struct dmzap_chunk_work *cw = container_of(work, struct dmzap_chunk_work, work);
366  struct dmzap_target *dmzap = cw->target;
367  struct bio *bio;
368 
369  mutex_lock(&dmzap->chunk_lock);
370 
371  /* Process the chunk BIOs */
372  while ((bio = bio_list_pop(&cw->bio_list))) {
373  mutex_unlock(&dmzap->chunk_lock);
374  dmzap_handle_bio(dmzap, cw, bio);
375  mutex_lock(&dmzap->chunk_lock);
376  dmzap_put_chunk_work(cw);
377  }
378 
379  /* Queueing the work incremented the work refcount */
380  dmzap_put_chunk_work(cw);
381 
382  mutex_unlock(&dmzap->chunk_lock);
383 }

409 /*
410  * Get a chunk work and start it to process a new BIO.
411  * If the BIO chunk has no work yet, create one.
412  */
413 static int dmzap_queue_chunk_work(struct dmzap_target *dmzap, struct bio *bio)
414 {
415  unsigned int chunk = dmzap_bio_chunk(dmzap->dev, bio);
416  struct dmzap_chunk_work *cw;
417  int ret = 0;
418 
419  mutex_lock(&dmzap->chunk_lock);
420 
421  /* Get the BIO chunk work. If one is not active yet, create one */
422  cw = radix_tree_lookup(&dmzap->chunk_rxtree, chunk);
423  if (!cw) {
424 
425  /* Create a new chunk work */
426  cw = kmalloc(sizeof(struct dmzap_chunk_work), GFP_NOIO);
427  if (unlikely(!cw)) {
428  ret = -ENOMEM;
429  goto out;
430  }
431 
432  INIT_WORK(&cw->work, dmzap_chunk_work_);
433  refcount_set(&cw->refcount, 1);
434  cw->target = dmzap;
435  cw->chunk = chunk;
436  bio_list_init(&cw->bio_list);
437 
438  ret = radix_tree_insert(&dmzap->chunk_rxtree, chunk, cw);
439  if (unlikely(ret)) {
440  kfree(cw);
441  goto out;
442  }
443  }
444 
445  bio_list_add(&cw->bio_list, bio);
446  dmzap_get_chunk_work(cw);
447 
448  dmzap_reclaim_bio_acc(dmzap);
449  if (queue_work(dmzap->chunk_wq, &cw->work))
450  dmzap_get_chunk_work(cw);
451 out:
452  mutex_unlock(&dmzap->chunk_lock);
453  return ret;
454 }

456 /*
457  * Check if the backing device is being removed. If it's on the way out,
458  * start failing I/O. Reclaim and metadata components also call this
459  * function to cleanly abort operation in the event of such failure.
460  */
461 bool dmzap_bdev_is_dying(struct dmz_dev *dmz_dev)
462 {
463  if (dmz_dev->flags & DMZ_BDEV_DYING)
464  return true;
465 
466  if (dmz_dev->flags & DMZ_CHECK_BDEV)
467  return !dmzap_check_bdev(dmz_dev);
468 
469  if (blk_queue_dying(bdev_get_queue(dmz_dev->bdev))) {
470  dmz_dev_warn(dmz_dev, "Backing device queue dying");
471  dmz_dev->flags |= DMZ_BDEV_DYING;
472  }
473 
474  return dmz_dev->flags & DMZ_BDEV_DYING;
475 }

477 /*
478  * Check the backing device availability. This detects such events as
479  * backing device going offline due to errors, media removals, etc.
480  * This check is less efficient than dmzap_bdev_is_dying() and should
481  * only be performed as a part of error handling.
482  */
483 bool dmzap_check_bdev(struct dmz_dev *dmz_dev)
484 {
485  struct gendisk *disk;
486 
487  dmz_dev->flags &= ~DMZ_CHECK_BDEV;
488 
489  if (dmzap_bdev_is_dying(dmz_dev))
490  return false;
491 
492  disk = dmz_dev->bdev->bd_disk;
493  if (disk->fops->check_events &&
494  disk->fops->check_events(disk, 0) & DISK_EVENT_MEDIA_CHANGE) {
495  dmz_dev_warn(dmz_dev, "Backing device offline");
496  dmz_dev->flags |= DMZ_BDEV_DYING;
497  }
498 
499  return !(dmz_dev->flags & DMZ_BDEV_DYING);
500 }

502 /*
503  * Process a bio
504  */
505 static int dmzap_map(struct dm_target *ti, struct bio *bio)
506 {
507  struct dmzap_target *dmzap = ti->private;
508  struct dmz_dev *dev = dmzap->dev;
509  sector_t sector = bio->bi_iter.bi_sector;
510  unsigned int nr_sectors = bio_sectors(bio);
511  sector_t chunk_sector;
512  int ret;
513 
514  if (dmzap_bdev_is_dying(dmzap->dev))
515  return DM_MAPIO_KILL;
516 
517  if(dmzap->show_debug_msg){
518  dmz_dev_debug(dev, "BIO op %d sector %llu + %u => chunk %llu, block %llu, %u blocks",
519  bio_op(bio), (unsigned long long)sector, nr_sectors,
520  (unsigned long long)dmzap_bio_chunk(dev, bio),
521  (unsigned long long)dmzap_chunk_block(dev, dmz_bio_block(bio)),
522  (unsigned int)dmz_bio_blocks(bio));
523  }
524 
525  bio_set_dev(bio, dev->bdev);
526 
527  if (!nr_sectors && bio_op(bio) != REQ_OP_WRITE)
528  return DM_MAPIO_REMAPPED;
529 
530  /* The BIO should be block aligned */
531  if ((nr_sectors & DMZ_BLOCK_SECTORS_MASK) || (sector & DMZ_BLOCK_SECTORS_MASK))
532  return DM_MAPIO_KILL;
533 
534  /* Initialize the BIO context */
535  dmzap_init_bioctx(dmzap,bio);
536 
537  /* Set the BIO pending in the flush list */
538  if (!nr_sectors && bio_op(bio) == REQ_OP_WRITE) {
539  spin_lock(&dmzap->flush_lock);
540  bio_list_add(&dmzap->flush_list, bio);
541  spin_unlock(&dmzap->flush_lock);
542  mod_delayed_work(dmzap->flush_wq, &dmzap->flush_work, 0);
543  return DM_MAPIO_SUBMITTED;
544  }
545 
546  /* Split zone BIOs to fit entirely into a zone */
547  chunk_sector = sector & (dev->zone_nr_sectors - 1);
548  if (chunk_sector + nr_sectors > dev->zone_nr_sectors)
549  dm_accept_partial_bio(bio, dev->zone_nr_sectors - chunk_sector);
550 
551  /* Now ready to handle this BIO */
552  ret = dmzap_queue_chunk_work(dmzap, bio);
553  if (ret) {
554  dmz_dev_debug(dev,
555  "BIO op %d, can't process chunk %llu, err %i\n",
556  bio_op(bio), (u64)dmzap_bio_chunk(dev, bio),
557  ret);
558  return DM_MAPIO_REQUEUE;
559  }
560 
561  return DM_MAPIO_SUBMITTED;
562 }

568 static int dmzap_get_zoned_device(struct dm_target *ti, char *path)
569 {
570  struct dmzap_target *dmz = ti->private;

572  struct dmz_dev *dev;

574  int ret;

578  if (ret) {
579  ti->error = "Get target device failed";
580  dmz->ddev = NULL;
581  return ret;
582  }
583 
584  dev = kzalloc(sizeof(struct dmz_dev), GFP_KERNEL);
585  if (!dev) {
586  ret = -ENOMEM;
587  goto err;
588  }

595  ret = -EINVAL;
596  goto err;
597  }
598

600  dev->capacity = i_size_read(dev->bdev->bd_inode) >> SECTOR_SHIFT;
