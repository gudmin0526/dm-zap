38  kvfree(map->d2l);

43  map->l2d[i] = DMZAP_UNMAPPED;

159 /*
160  * Look up mapped blocks, returns:
161  * number of contigous sectors (<= len)
162  */
163 int dmzap_map_lookup(struct dmzap_target *dmzap,
164  sector_t user, sector_t *backing, sector_t len)
165 {
166  struct dmzap_map *map = &dmzap->map;
167  sector_t m = user;
168  sector_t left = len - 1;
169 
170  /* Out of bounds? */
171  if ((user + len) > dmz_sect2blk(dmzap->capacity)) {
172  BUG();
173  return -1;
174  }
175 
176  *backing = map->l2d[user];
177 
178  if (map->l2d[user] == DMZAP_UNMAPPED) {
179  *backing = DMZAP_UNMAPPED;
180  while (left && (map->l2d[m] == DMZAP_UNMAPPED)) {
181  m++;
182  left--;
183  }
184  } else if(map->l2d[user] >= 0 && map->invalid_device_block[map->l2d[user]]){
185  *backing = DMZAP_INVALID;
186  while (left && (map->l2d[m] >= 0
187  && map->invalid_device_block[map->l2d[m]])) {
188  m++;
189  left--;
190  }
191  } else {
192  while (left && ((map->l2d[m] + 1) == map->l2d[m+1])
193  && !map->invalid_device_block[map->l2d[m]] && !map->invalid_device_block[map->l2d[m+1]] ) {
194  m++;
195  left--;
196  }
197  }
198 
199  if(dmzap->show_debug_msg){
200  dmz_dev_debug(dmzap->dev, "looked up %d user block(s) from %d to backing block: %d",
201  (int)(len - left), (int)user, (int)*backing);
202  }
203 
204  return len - left;
205 }
